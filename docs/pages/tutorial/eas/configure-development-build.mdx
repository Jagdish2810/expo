My Dairy - Starter App (Expo React Native) + Admin Web Panel (React + Vite)
=======================================================================

Contents:
- mobile/        -> Expo React Native starter (App.js, screens, components)
- admin/         -> React (Vite) admin panel starter (src, vite config)
- docs/          -> run/build instructions, env examples
- assets/        -> placeholder logo and images
- LICENSE        -> MIT

What I built for you:
- Minimal, well-structured starter code for the My Dairy mobile app (Expo) and an Admin web panel (React + Vite).
- Auth placeholders for roles, sample screens: Login, Farmer Dashboard, Scan, Register Animal, Milk Report, Activate Tag.
- Admin panel skeleton: login, dashboard, farmer list, tag management, reports page.
- A README with commands to run the projects locally and how to build an Android APK using Expo.
- Notes on next steps, backend suggestions, and recommended libraries.

How to run (locally):
1) Mobile (Expo)
   - Install expo-cli if you don't have it: npm install -g expo-cli
   - cd mobile
   - npm install
   - expo start
   - Use Expo Go on Android/iOS to test or build a standalone APK with 'eas build' (recommended) or 'expo build:android' (classic).

2) Admin (React, Vite)
   - cd admin
   - npm install
   - npm run dev
   - Open http://localhost:5173

Notes:
- Backend API is NOT included. The mobile and admin projects use placeholder/mock data and a simple `./src/config/api.js` where you can plug your real backend endpoints.
- For production, set up a REST API (Node/Express, Django, or Firebase) and a database (Postgres/MySQL/MongoDB).

""")

ScanScreen.js
import React, { useState, useEffect } from 'react';
import { View, StyleSheet, Text } from 'react-native';
import { BarCodeScanner } from 'expo-barcode-scanner';

export default function ScanScreen() {
  const [hasPermission, setHasPermission] = useState(null);
  const [scanned, setScanned] = useState(false);

  useEffect(() => {
    (async () => {
      const { status } = await BarCodeScanner.requestPermissionsAsync();
      setHasPermission(status === 'granted');
    })();
  }, []);

  const handleBarCodeScanned = ({ type, data }) => {
    setScanned(true);
    alert(`Ear Tag Scanned: ${data}`);
  };

  if (hasPermission === null) {
    return <Text>Requesting camera permission...</Text>;
  }
  if (hasPermission === false) {
    return <Text>No access to camera</Text>;
  }

  return (
    <View style={styles.container}>
      <BarCodeScanner
        onBarCodeScanned={scanned ? undefined : handleBarCodeScanned}
        style={StyleSheet.absoluteFillObject}
      />
      {scanned && <Text style={styles.scanText}>Scanned! Tap back to rescan.</Text>}
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1 },
  scanText: { position: 'absolute', bottom: 30, alignSelf: 'center', color: '#fff', backgroundColor:'#0008', padding:8 }
});

MilkReport.js
import React, { useState } from 'react';
import { View, StyleSheet } from 'react-native';
import { TextInput, Button, Text } from 'react-native-paper';

export default function MilkReport() {
  const [morning, setMorning] = useState('');
  const [evening, setEvening] = useState('');
  const [records, setRecords] = useState([]);

  const addRecord = () => {
    const total = (parseFloat(morning) || 0) + (parseFloat(evening) || 0);
    const newRecord = { morning, evening, total, date: new Date().toLocaleDateString() };
    setRecords([newRecord, ...records]);
    setMorning(''); setEvening('');
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Milk Report</Text>
      <TextInput label="Morning (litres)" value={morning} onChangeText={setMorning} keyboardType="numeric" style={styles.input} />
      <TextInput label="Evening (litres)" value={evening} onChangeText={setEvening} keyboardType="numeric" style={styles.input} />
      <Button mode="contained" onPress={addRecord}>Add Record</Button>

      {records.map((r, i) => (
        <View key={i} style={styles.record}>
          <Text>{r.date}: {r.total} L (M:{r.morning}, E:{r.evening})</Text>
        </View>
      ))}
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: '#fff', padding: 16 },
  title: { fontSize: 20, fontWeight: '700', marginBottom: 16 },
  input: { marginBottom: 12 },
  record: { marginTop: 10, padding: 8, backgroundColor:'#f5f5f5', borderRadius:6 }
});

import React, { useState, useEffect } from 'react';
import { View, StyleSheet, Text } from 'react-native';
import { BarCodeScanner } from 'expo-barcode-scanner';

export default function ScanScreen() {
  const [hasPermission, setHasPermission] = useState(null);
  const [scanned, setScanned] = useState(false);

  useEffect(() => {
    (async () => {
      const { status } = await BarCodeScanner.requestPermissionsAsync();
      setHasPermission(status === 'granted');
    })();
  }, []);

  const handleBarCodeScanned = ({ type, data }) => {
    setScanned(true);
    alert(`Ear Tag Scanned: ${data}`);
  };

  if (hasPermission === null) {
    return <Text>Requesting camera permission...</Text>;
  }
  if (hasPermission === false) {
    return <Text>No access to camera</Text>;
  }

  return (
    <View style={styles.container}>
      <BarCodeScanner
        onBarCodeScanned={scanned ? undefined : handleBarCodeScanned}
        style={StyleSheet.absoluteFillObject}
      />
      {scanned && <Text style={styles.scanText}>Scanned! Tap back to rescan.</Text>}
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1 },
  scanText: { position: 'absolute', bottom: 30, alignSelf: 'center', color: '#fff', backgroundColor:'#0008', padding:8 }
});

import React, { useState } from 'react';
import { View, StyleSheet } from 'react-native';
import { TextInput, Button, Text } from 'react-native-paper';

export default function MilkReport() {
  const [morning, setMorning] = useState('');
  const [evening, setEvening] = useState('');
  const [records, setRecords] = useState([]);

  const addRecord = () => {
    const total = (parseFloat(morning) || 0) + (parseFloat(evening) || 0);
    const newRecord = { morning, evening, total, date: new Date().toLocaleDateString() };
    setRecords([newRecord, ...records]);
    setMorning(''); setEvening('');
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Milk Report</Text>
      <TextInput label="Morning (litres)" value={morning} onChangeText={setMorning} keyboardType="numeric" style={styles.input} />
      <TextInput label="Evening (litres)" value={evening} onChangeText={setEvening} keyboardType="numeric" style={styles.input} />
      <Button mode="contained" onPress={addRecord}>Add Record</Button>

      {records.map((r, i) => (
        <View key={i} style={styles.record}>
          <Text>{r.date}: {r.total} L (M:{r.morning}, E:{r.evening})</Text>
        </View>
      ))}
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: '#fff', padding: 16 },
  title: { fontSize: 20, fontWeight: '700', marginBottom: 16 },
  input: { marginBottom: 12 },
  record: { marginTop: 10, padding: 8, backgroundColor:'#f5f5f5', borderRadius:6 }
});
