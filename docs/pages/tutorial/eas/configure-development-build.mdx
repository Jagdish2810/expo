export const API_BASE = "http://localhost:4000/api"; // replace with server URL

export async function registerAnimal(data) {
  const res = await fetch(`${API_BASE}/animals`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(data)
  });
  return res.json();
}

export async function addMilkRecord(animalId, data) {
  const res = await fetch(`${API_BASE}/animals/${animalId}/milk`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(data)
  });
  return res.json();
}

const express = require("express");
const cors = require("cors");
const bodyParser = require("body-parser");

const app = express();
app.use(cors());
app.use(bodyParser.json());

// --- In-Memory Data Store ---
let farmers = [];
let animals = [];
let milkRecords = [];
let healthReports = [];
let vaccinations = [];
let animalTrades = [];
let payments = [];
let dairies = [];
let feedSuppliers = [];
let doctors = [];

// ========== FARMERS ==========
app.get("/api/farmers", (req, res) => res.json(farmers));

app.post("/api/farmers", (req, res) => {
  const { id, name, village } = req.body;
  const farmer = { id, name, village, animals: [] };
  farmers.push(farmer);
  res.json(farmer);
});

// ========== ANIMALS ==========
app.get("/api/animals", (req, res) => res.json(animals));

app.post("/api/animals", (req, res) => {
  const { tag, breed, age, gender, farmerId } = req.body;
  const animal = { tag, breed, age, gender, farmerId, milk: [] };
  animals.push(animal);

  const farmer = farmers.find(f => f.id === farmerId);
  if (farmer) farmer.animals.push(animal);

  res.json(animal);
});

// ========== MILK ==========
app.post("/api/animals/:tag/milk", (req, res) => {
  const { tag } = req.params;
  const { litres, date } = req.body;

  const animal = animals.find(a => a.tag === tag);
  if (!animal) return res.status(404).json({ error: "Animal not found" });

  const record = { tag, litres, date: date || new Date().toISOString() };
  animal.milk.push(record);
  milkRecords.push(record);

  res.json(record);
});

app.get("/api/reports/milk", (req, res) => {
  const summary = {};
  milkRecords.forEach(r => {
    const d = r.date.split("T")[0];
    if (!summary[d]) summary[d] = { date: d, total: 0, count: 0 };
    summary[d].total += r.litres;
    summary[d].count++;
  });
  res.json(Object.values(summary));
});

// ========== HEALTH ==========
app.post("/api/animals/:tag/health", (req, res) => {
  const { tag } = req.params;
  const { status, notes, date } = req.body;
  const report = { tag, status, notes, date: date || new Date().toISOString() };
  healthReports.push(report);
  res.json(report);
});

app.get("/api/reports/health/:tag", (req, res) => {
  const { tag } = req.params;
  res.json(healthReports.filter(r => r.tag === tag));
});

// ========== VACCINATION ==========
app.post("/api/animals/:tag/vaccination", (req, res) => {
  const { tag } = req.params;
  const { vaccine, doctor, date } = req.body;
  const record = { tag, vaccine, doctor, date: date || new Date().toISOString() };
  vaccinations.push(record);
  res.json(record);
});

app.get("/api/reports/vaccination/:tag", (req, res) => {
  const { tag } = req.params;
  res.json(vaccinations.filter(v => v.tag === tag));
});

// ========== ANIMAL TRADE ==========
app.post("/api/trade", (req, res) => {
  const { sellerId, buyerId, tag, price, date } = req.body;
  const trade = { sellerId, buyerId, tag, price, date: date || new Date().toISOString() };
  animalTrades.push(trade);
  res.json(trade);
});

app.get("/api/trade", (req, res) => res.json(animalTrades));

// ========== PAYMENTS ==========
app.post("/api/payments", (req, res) => {
  const { from, to, amount, method, date } = req.body;
  const payment = { from, to, amount, method, date: date || new Date().toISOString() };
  payments.push(payment);
  res.json(payment);
});

app.get("/api/payments", (req, res) => res.json(payments));

// ========== LOCAL DAIRIES ==========
app.get("/api/dairies", (req, res) => res.json(dairies));

app.post("/api/dairies", (req, res) => {
  const { id, name, location } = req.body;
  const dairy = { id, name, location };
  dairies.push(dairy);
  res.json(dairy);
});

// ========== FEED SUPPLIERS ==========
app.get("/api/feedsuppliers", (req, res) => res.json(feedSuppliers));

app.post("/api/feedsuppliers", (req, res) => {
  const { id, name, products } = req.body;
  const supplier = { id, name, products };
  feedSuppliers.push(supplier);
  res.json(supplier);
});

// ========== DOCTORS ==========
app.get("/api/doctors", (req, res) => res.json(doctors));

app.post("/api/doctors", (req, res) => {
  const { id, name, specialization } = req.body;
  const doctor = { id, name, specialization };
  doctors.push(doctor);
  res.json(doctor);
});

// ========== SERVER ==========
const PORT = 4000;
app.listen(PORT, () => {
  console.log(`🚀 My Dairy extended backend running at http://localhost:${PORT}`);
});

import matplotlib.pyplot as plt
import numpy as np
from pptx import Presentation
from pptx.util import Inches, Pt

def create_role_dashboard(role_index, role_name, benefits, color, filename_prefix):
    # Define center
    center_x, center_y = 0, 0

    # Define outer roles (7 roles)
    roles = [
        "👨‍🌾 Farmer",
        "🥛 Milk Trader",
        "🏭 Local Dairy",
        "🐂 Animal Seller",
        "🌾 Feed Seller",
        "👨‍⚕️ Vet Doctor",
        "🏦 Insurance"
    ]

    # Colors for roles
    colors = ["#7ED957", "#FFB347", "#4A90E2", "#FF6961", "#FDD835", "#BA68C8", "#00ACC1"]

    # Angles for radial placement
    angles = np.linspace(0, 2*np.pi, len(roles), endpoint=False)
    radius = 6

    # Plot setup
    fig, ax = plt.subplots(figsize=(12,12))
    ax.set_xlim(-9, 9)
    ax.set_ylim(-9, 9)
    ax.axis("off")
    fig.patch.set_facecolor("white")

    # Draw center circle (Selected Role)
    circle = plt.Circle((center_x, center_y), 1.5, color=color, ec="black", lw=2)
    ax.add_patch(circle)
    ax.text(center_x, center_y, role_name + "\n(Dashboard)", ha="center", va="center",
            fontsize=10, color="black", weight="bold", wrap=True)

    # Draw outer roles
    for i, r in enumerate(roles):
        if i == role_index:
            continue  # skip center role
       
        x = radius * np.cos(angles[i])
        y = radius * np.sin(angles[i])

        # Node circle with colors
        outer_circle = plt.Circle((x, y), 1.5, color=colors[i], ec="black", lw=1.5)
        ax.add_patch(outer_circle)

        # Role text
        ax.text(x, y, r, ha="center", va="center", fontsize=9, wrap=True)

        # Connecting arrow
        ax.annotate("", xy=(x*0.8, y*0.8), xytext=(center_x, center_y),
                    arrowprops=dict(arrowstyle="<->", lw=1.2, color="black"))

    plt.title(f"My Dairy – {role_name} Dashboard\nSkyplanet Livestock Care", fontsize=14, weight="bold")

    # Save PNG
    png_file = f"/mnt/data/{filename_prefix}_Dashboard.png"
    plt.savefig(png_file, dpi=300, bbox_inches="tight")
    plt.close()

    # Generate PPTX
    pptx_file = f"/mnt/data/{filename_prefix}_Dashboard.pptx"
    prs = Presentation()
    slide_layout = prs.slide_layouts[5]
    slide = prs.slides.add_slide(slide_layout)

    # Title
    title = slide.shapes.title
    title.text = f"My Dairy – {role_name} Dashboard\nSkyplanet Livestock Care"
    text_frame = title.text_frame
    text_frame.paragraphs[0].font.size = Pt(28)
    text_frame.paragraphs[0].font.bold = True

    # Insert PNG
    left = Inches(0.5)
    top = Inches(1.5)
    slide.shapes.add_picture(png_file, left, top, width=Inches(5.5))

    # Add Benefits box
    left = Inches(6.5)
    top = Inches(1.5)
    width = Inches(3)
    height = Inches(4)
    txBox = slide.shapes.add_textbox(left, top, width, height)
    tf = txBox.text_frame
    tf.word_wrap = True
    p = tf.add_paragraph()
    p.text = f"{role_name} Benefits:"
    p.font.size = Pt(20)
    p.font.bold = True

    for b in benefits:
        p = tf.add_paragraph()
        p.text = "• " + b
        p.font.size = Pt(16)

    prs.save(pptx_file)
    return png_file, pptx_file

import React, { useState, useEffect } from 'react';
import { View, StyleSheet, Text } from 'react-native';
import { BarCodeScanner } from 'expo-barcode-scanner';

export default function ScanScreen() {
  const [hasPermission, setHasPermission] = useState(null);
  const [scanned, setScanned] = useState(false);

  useEffect(() => {
    (async () => {
      const { status } = await BarCodeScanner.requestPermissionsAsync();
      setHasPermission(status === 'granted');
    })();
  }, []);

  const handleBarCodeScanned = ({ type, data }) => {
    setScanned(true);
    alert(`Ear Tag Scanned: ${data}`);
  };

  if (hasPermission === null) {
    return <Text>Requesting camera permission...</Text>;
  }
  if (hasPermission === false) {
    return <Text>No access to camera</Text>;
  }

  return (
    <View style={styles.container}>
      <BarCodeScanner
        onBarCodeScanned={scanned ? undefined : handleBarCodeScanned}
        style={StyleSheet.absoluteFillObject}
      />
      {scanned && <Text style={styles.scanText}>Scanned! Tap back to rescan.</Text>}
    </View>
  );
}

const styles = StyleSheet.create({

  scanText: { position: 'absolute', bottom: 30, alignSelf: 'center', color: '#fff', backgroundColor:'#0008', padding:8 }

import React, { useState } from 'react';
import { View, StyleSheet } from 'react-native';
import { TextInput, Button, Text } from 'react-native-paper';

export default function MilkReport() {
  const [morning, setMorning] = useState('');
  const [evening, setEvening] = useState('');
  const [records, setRecords] = useState([]);

  const addRecord = () => {
    const total = (parseFloat(morning) || 0) + (parseFloat(evening) || 0);
    const newRecord = { morning, evening, total, date: new Date().toLocaleDateString() };
    setRecords([newRecord, ...records]);
    setMorning(''); setEvening('');
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Milk Report</Text>
      <TextInput label="Morning (litres)" value={morning} onChangeText={setMorning} keyboardType="numeric" style={styles.input} />
      <TextInput label="Evening (litres)" value={evening} onChangeText={setEvening} keyboardType="numeric" style={styles.input} />
      <Button mode="contained" onPress={addRecord}>Add Record</Button>

      {records.map((r, i) => (
        <View key={i} style={styles.record}>
          <Text>{r.date}: {r.total} L (M:{r.morning}, E:{r.evening})</Text>
        </View>
      ))}
    </View>
  );
}

import React, { useState, useEffect } from 'react';
import { View, StyleSheet, Text } from 'react-native';
import { BarCodeScanner } from 'expo-barcode-scanner';

export default function ScanScreen() {
  const [hasPermission, setHasPermission] = useState(null);
  const [scanned, setScanned] = useState(false);

  useEffect(() => {
    (async () => {
      const { status } = await BarCodeScanner.requestPermissionsAsync();
      setHasPermission(status === 'granted');
    })();
  }, []);

  const handleBarCodeScanned = ({ type, data }) => {
    setScanned(true);
    alert(`Ear Tag Scanned: ${data}`);
  };

  if (hasPermission === null) {
    return <Text>Requesting camera permission...</Text>;
  }
  if (hasPermission === false) {
    return <Text>No access to camera</Text>;
  }

  return (
    <View style={styles.container}>
      <BarCodeScanner
        onBarCodeScanned={scanned ? undefined : handleBarCodeScanned}
        style={StyleSheet.absoluteFillObject}
      />
      {scanned && <Text style={styles.scanText}>Scanned! Tap back to rescan.</Text>}
    </View>
  );
}

const styles = StyleSheet.create({

import React, { useState } from 'react';

export default function MilkReport() {
  const [morning, setMorning] = useState('');
  const [evening, setEvening] = useState('');
  const [records, setRecords] = useState([]);

  const addRecord = () => {
    const total = (parseFloat(morning) || 0) + (parseFloat(evening) || 0);
    const newRecord = { morning, evening, total, date: new Date().toLocaleDateString() };
    setRecords([newRecord, ...records]);
    setMorning(''); setEvening('');
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Milk Report</Text>
      <TextInput label="Morning (litres)" value={morning} onChangeText={setMorning} keyboardType="numeric" style={styles.input} />
      <TextInput label="Evening (litres)" value={evening} onChangeText={setEvening} keyboardType="numeric" style={styles.input} />
      <Button mode="contained" onPress={addRecord}>Add Record</Button>

      {records.map((r, i) => (
        <View key={i} style={styles.record}>
          <Text>{r.date}: {r.total} L (M:{r.morning}, E:{r.evening})</Text>
        </View>
      ))}
    </View>
  );
}
